## Coding — Text Classification Substring Problem

Find the longest contiguous substring of identical characters;
if multiple of equal length, pick the rightmost one.
Return a string like "c3".
```
def solution(source):
    max_len = 0
    max_char = ''
    curr_len = 1
    for i in range(1, len(source)):
        if source[i] == source[i-1]:
            curr_len += 1
        else:
            curr_len = 1
        if curr_len >= max_len:
            max_len = curr_len
            max_char = source[i]
    return f"{max_char}{max_len}"
```

## Coding — Implementing Gradient Descent for Linear Regression

You’re asked to implement gradient descent from scratch for the model
```
import random

def random_bias_theta(n_features: int):
    """STEP 1: Initialize bias and theta"""
    b = random.random()
    theta = [random.random() for _ in range(n_features)]
    return b, theta


def calculate_y(b: float, theta: list[float], x: list[float]) -> float:
    """STEP 2: Predicted value of y"""
    return b + sum(t * xi for t, xi in zip(theta, x))


def calculate_d_b(Y: list[float], Y_hat: list[float]) -> float:
    """STEP 3: Partial derivative wrt bias"""
    n = len(Y)
    return (-2 / n) * sum(y - y_hat for y, y_hat in zip(Y, Y_hat))


def calculate_d_theta(X: list[list[float]], Y: list[float], Y_hat: list[float]) -> list[float]:
    """STEP 4: Partial derivative wrt theta"""
    n = len(Y)
    m = len(X[0])
    d_theta = []
    for j in range(m):
        grad = (-2 / n) * sum(X[i][j] * (Y[i] - Y_hat[i]) for i in range(n))
        d_theta.append(grad)
    return d_theta


def update(X, Y, Y_hat, b_prev, theta_prev, learning_rate):
    """STEP 5: Update gradient and weights"""
    d_b = calculate_d_b(Y, Y_hat)
    d_theta = calculate_d_theta(X, Y, Y_hat)
    b_new = b_prev - learning_rate * d_b
    theta_new = [theta_prev[j] - learning_rate * d_theta[j] for j in range(len(theta_prev))]
    return b_new, theta_new


def fit(X, Y, num_iterations=1000, learning_rate=0.2):
    """STEP 6: Run gradient descent loop"""
    b, theta = random_bias_theta(len(X[0]))
    for _ in range(num_iterations):
        Y_hat = [calculate_y(b, theta, x) for x in X]
        b, theta = update(X, Y, Y_hat, b, theta, learning_rate)
    return b, theta


def solution(x_train, y_train, x_test, iterations=1000):
    random.seed(42)
    b, theta = fit(x_train, y_train, iterations)
    return [round(calculate_y(b, theta, x), 2) for x in x_test]
```

## Coding — Implementing k-Means Clustering

Given:data: list of 2D points

k: number of clusters

centroids: initial centroids

iterations: number of iterations

Algorithm steps:

Compute the distance from each data point to each centroid
→ Euclidean distance

Assign each point to the nearest centroid (lowest distance)

Update centroids as the mean of all assigned points

Repeat steps 1–3 for the given iterations

```
def solution(data, k, centroids, iterations):
    for _ in range(iterations):
        # STEP 1: Assign each point to the nearest centroid
        clusters = [[] for _ in range(k)]
        for point in data:
            distances = [((point[0]-c[0])**2 + (point[1]-c[1])**2)**0.5 for c in centroids]
            cluster_idx = distances.index(min(distances))
            clusters[cluster_idx].append(point)
        
        # STEP 2: Update centroids
        for i in range(k):
            if clusters[i]:  # avoid empty clusters
                x_mean = sum(p[0] for p in clusters[i]) / len(clusters[i])
                y_mean = sum(p[1] for p in clusters[i]) / len(clusters[i])
                centroids[i] = [x_mean, y_mean]
    
    # STEP 3: Assign each point to the final cluster
    labels = []
    for point in data:
        distances = [((point[0]-c[0])**2 + (point[1]-c[1])**2)**0.5 for c in centroids]
        labels.append(distances.index(min(distances)))
    return labels
```

```
from scipy.spatial.distance import cdist
from typing import List

def calculate_distances(data: List[List[float]], centroids: List[List[float]]) -> List[List[float]]:
    """
    Step 1: Calculate distance between each data point and the k centroids
    """
    return cdist(data, centroids, 'euclidean').tolist()


def get_clusters(data: List[List[float]], centroids: List[List[float]]) -> List[int]:
    """
    Step 2: Assign each data point to its nearest centroid
    """
    distances = calculate_distances(data, centroids)
    clusters = [row.index(min(row)) for row in distances]
    return clusters


def update_clusters(clusters: List[int], data: List[List[float]], k: int) -> List[List[float]]:
    """
    Step 3: Average the data points in each cluster to update the centroids
    """
    new_centroids = []
    for i in range(k):
        # Extract points belonging to cluster i
        cluster_points = [data[j] for j in range(len(data)) if clusters[j] == i]
        if cluster_points:
            # Compute mean for each dimension
            new_centroid = [sum(dim) / len(cluster_points) for dim in zip(*cluster_points)]
        else:
            # If a cluster is empty, keep the old centroid as placeholder
            new_centroid = [0.0 for _ in range(len(data[0]))]
        new_centroids.append(new_centroid)
    return new_centroids


def fit_predict(data: List[List[float]], k: int, centroids: List[List[float]], iterations: int) -> List[int]:
    """
    Step 4: Pull everything together
    """
    for _ in range(iterations):
        clusters = get_clusters(data, centroids)
        centroids = update_clusters(clusters, data, k)
    return get_clusters(data, centroids)


def solution(data: List[List[float]], k: int, centroids: List[List[float]], iterations: int) -> List[int]:
    return fit_predict(data, k, centroids, iterations)
```

✅ Explanation

calculate_distances
Uses Euclidean distance (via cdist) to compute a matrix where each entry (i, j) is the distance from data point i to centroid j.

get_clusters
For each data point (row), find the index of the smallest distance — the cluster ID.

update_clusters
Recomputes each centroid as the mean of all points assigned to that cluster.
If a cluster is empty, it returns [0.0, 0.0, ...] for stability (though that rarely occurs with provided constraints).

fit_predict
Iteratively reassigns points and updates centroids for the given number of iterations.
After the last update, returns the final cluster assignments.
